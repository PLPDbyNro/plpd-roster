window.deleteOfficer = async function(id) { if (confirm('Are you sure you want to delete this officer record?')) { try { const response = await fetch(`${API_URL}/${id}`, { method: 'DELETE' }); if (response.ok) { await fetchLiveRoster(); } else { officers = officers.filter(o => String(o.id) !== String(id)); renderRoster(); } } catch (err) { console.warn('Backend delete failed, performing local UI update', err); officers = officers.filter(o => String(o.id) !== String(id)); renderRoster(); } } }; // Export CSV functionality document.getElementById('export-csv').addEventListener('click', () => { let csvContent = "data:text/csv;charset=utf-8,Badge,Name,Rank,Responsibility,Status,Strikes,Discord ID\n"; officers.forEach(o => { const parsed = processOfficerBadgeAndName(o.name, o.badge); let row = [parsed.badge, `"${parsed.name}"`, `"${o.rank}"`, `"${o.responsibility}"`, o.status || 'Active', o.strikes || 0, o.discord || 'N/A']; csvContent += row.join(",") + "\n"; }); const encodedUri = encodeURI(csvContent); const link = document.createElement("a"); link.setAttribute("href", encodedUri); link.setAttribute("download", "police_department_roster.csv"); document.body.appendChild(link); link.click(); document.body.removeChild(link); }); // Initial fetch on page load fetchLiveRoster(); // Auto-refresh every 30 seconds setInterval(fetchLiveRoster, 30000);
